# EJEMPLO: Configuración con IP real
# Supongamos que tu servidor tiene IP: 203.0.113.45

# 1. En el servidor, verificar la IP pública:
curl ifconfig.me
# Resultado: 203.0.113.45

# 2. Probar que el backend funciona:
curl http://203.0.113.45:3000/health
# Debe devolver: {"status":"ok","timestamp":"..."}

# 3. En la app, actualizar config/backend.ts:

# ANTES:
return 'http://TU_IP_PUBLICA_AQUI:3000';

# DESPUÉS:
return 'http://203.0.113.45:3000';

# 4. También actualizar la configuración de producción:

# ANTES:
production: {
  baseUrl: 'http://TU_IP_PUBLICA_AQUI:3000',
  ...
}

# DESPUÉS:
production: {
  baseUrl: 'http://203.0.113.45:3000',
  ...
}
